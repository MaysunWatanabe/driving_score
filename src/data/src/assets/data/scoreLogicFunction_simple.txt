// 交差点名を取得する関数
const isIntersectionFunc = (geolocation, intersectionList) => {
  //小数点以下4桁 (精度7) は8m以内
  const accuracy = 10000;
  const lat = Math.round(geolocation.latitude * accuracy);
  const lon = Math.round(geolocation.longitude * accuracy);

  for (let i=0; i<intersectionList.length; i++) {
    const intersection = intersectionList[i];
    const lat2 = Math.round(intersection.lat * accuracy);
    const lon2 = Math.round(intersection.lon * accuracy);
    if (lat == lat2 && lon == lon2 && intersection.checked === undefined) {
      log.debug('[DrivingScore][ScoreLogic] intersection=' + intersection.name);
      return intersection.name;
    }
  }
  return '';
};

//メッセージを取得する関数
const getMessage = (json, key, area, score) => {
  //運転診断用のメッセージ
  for (let i=0; i<json.messages.length; i++) {
    if (json.messages[i].key == key
      && json.messages[i].area == area
      && json.messages[i].score.inclusive_min <= score
      && (json.messages[i].score.exclusive_max === undefined || score < json.messages[i].score.exclusive_max)
    ) {
      return json.messages[i];
    }
  }
  //能力指標用のメッセージ
  for (let i=0; i<json.capability_messages.length; i++) {
    if (json.capability_messages[i].key == key
      && json.capability_messages[i].area == area
      && json.capability_messages[i].score.inclusive_min <= score
      && (json.capability_messages[i].score.exclusive_max === undefined || score < json.capability_messages[i].score.exclusive_max)
    ) {
      return json.capability_messages[i];
    }
  }
  return null;
};

let speedMax = 0;

  // 前回実行時から増えたセンサー値から一番大きい取得

    speedMax = Math.max(speedMax , sensor.speed ?? 0);

  // 前回実行時から増えたセンサー値から一番大きい/小さい値を取得
  if (lastTimestamp < sensor.timestamp) {
    accelMaxZ = Math.max(accelMaxZ, sensor.accelerationIncludingGravity.rotate.lowPass.z);
    accelMinZ = Math.min(accelMinZ, sensor.accelerationIncludingGravity.rotate.lowPass.z);
  }
});
// ジャイロセンサー
let gammaMax = 0;
let gammaMin = 0;
gyroscopeList.forEach((sensor) => {
  if (lastTimestamp < sensor.timestamp) {
    gammaMax = Math.max(gammaMax, sensor.rotate.gamma);
    gammaMin = Math.min(gammaMin, sensor.rotate.gamma);
  }
});

if (localData.negativeTime === undefined) {
